% Caja de texto cifrado con su medidor de ruido encima.
% #1 nombre del nodo, #2 posicion, #3 contenido, #4 fraccion de ruido (0..1).
% Medidor de ruido horizontal sobre un nodo ya colocado. #1 nodo, #2 fraccion (0..1).
\newcommand{\medidor}[2]{%
    \coordinate (#1-gl) at ([yshift=3pt]#1.north west);
    \coordinate (#1-gr) at ([yshift=6pt]#1.north east);
    \fill[UCVOrange!85] (#1-gl) rectangle ($(#1-gl |- #1-gr)!#2!(#1-gr)$);
    \draw[gray!55, thin] (#1-gl) rectangle (#1-gr);
}
% Marca del umbral a partir del cual el texto cifrado deja de ser descifrable.
\newcommand{\umbral}[1]{%
    \draw[red!65!black, line width=0.6pt]
        ($(#1-gl)!0.93!(#1-gl -| #1-gr)$) -- ($(#1-gl |- #1-gr)!0.93!(#1-gr)$);
}
% Candado en la esquina inferior derecha, del color de la clave. #1 nodo, #2 color.
\newcommand{\candado}[2]{%
    \node[fill=white, inner sep=0.4pt, font=\tiny, text=#2] at (#1.south east) {\faLock};
}

\newcommand{\cajacifrada}[4]{%
    \node[cifrada] (#1) at #2 {#3};
    \node[fill=white, inner sep=0.4pt, font=\tiny, text=UCVBlue]
        at (#1.south east) {\faLock};
    \coordinate (#1-gl) at ([yshift=3pt]#1.north west);
    \coordinate (#1-gr) at ([yshift=6pt]#1.north east);
    \fill[UCVOrange!85] (#1-gl) rectangle ($(#1-gl |- #1-gr)!#4!(#1-gr)$);
    \draw[gray!55, thin] (#1-gl) rectangle (#1-gr);
}

\begin{frame}{FHE y el ruido}
    La seguridad de FHE descansa en el problema de \textbf{aprendizaje con errores}: cada texto cifrado lleva incorporado un pequeño ruido aleatorio que lo hace indescifrable. Ese ruido \textbf{se acumula con cada operación}.
    \vfill
    % Sube el bloque (diagrama y texto) respecto al fondo de la lamina.
    \vspace*{-0.5cm}
    \begin{columns}[t]
        \column{0.56\textwidth}
        \begin{center}
            \begin{tikzpicture}[every node/.style={font=\footnotesize},
                cifrada/.style={draw=UCVBlue, fill=UCVBlue!8, rounded corners=3pt,
                                minimum width=1.3cm, minimum height=0.9cm},
                op/.style={font=\normalsize, text=UCVOrange},
                igual/.style={font=\normalsize, text=UCVBlue}]

                % Suma: el ruido de los operandos apenas se acumula.
                \cajacifrada{sx}{(0,0.85)}{$x$}{0.18}
                \node[op] at (1.28,0.85) {$+$};
                \cajacifrada{sy}{(2.56,0.85)}{$y$}{0.18}
                \node[igual] at (3.84,0.85) {$=$};
                \cajacifrada{sr}{(5.12,0.85)}{$x\,\textcolor{UCVOrange}{+}\,y$}{0.34}

                % Multiplicacion: el ruido crece mucho mas.
                \cajacifrada{mx}{(0,-0.95)}{$x$}{0.18}
                \node[op] at (1.28,-0.95) {$\times$};
                \cajacifrada{my}{(2.56,-0.95)}{$y$}{0.18}
                \node[igual] at (3.84,-0.95) {$=$};
                \cajacifrada{mr}{(5.12,-0.95)}{$x\,\textcolor{UCVOrange}{\times}\,y$}{0.82}

            \end{tikzpicture}
        \end{center}
        \column{0.44\textwidth}
        \begin{itemize}
            \setlength{\itemsep}{8pt}
            \item \textbf{Suma:} el ruido crece de forma lineal.
            \item \textbf{Multiplicación:} el ruido crece de forma exponencial.
            \item Si el ruido supera un umbral, el texto cifrado \textbf{deja de ser descifrable}.
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Cifrado por niveles}
    Los esquemas de FHE modernos usan \textbf{cifrado por niveles}. Esto consiste en que se tiene una profundidad de cómputo predefinida y cada multiplicación consume un nivel. Al agotarse el último, el texto cifrado ya no admite más multiplicaciones.
    \vfill
    \begin{center}
        \begin{tikzpicture}[every node/.style={font=\scriptsize},
            lvl/.style={draw=UCVBlue, fill=UCVBlue!12, rounded corners=3pt,
                        minimum width=1.8cm, minimum height=0.85cm}]
            \node[lvl] (L3) at (0,0)   {$Q_L$ (inicio)};
            \node[lvl] (L2) at (2.75,0) {$Q_{L-1}$};
            \node[lvl] (L1) at (5.5,0) {$\dots$};
            \node[lvl, draw=UCVOrange, fill=UCVOrange!22] (L0) at (8.25,0) {$Q_0$ (límite)};

            \draw[->, thick, UCVOrange] (L3)
                -- node[above, font=\scriptsize, text=UCVOrange] {$\times$} (L2);
            \draw[->, thick, UCVOrange] (L2)
                -- node[above, font=\scriptsize, text=UCVOrange] {$\times$} (L1);
            \draw[->, thick, UCVOrange] (L1)
                -- node[above, font=\scriptsize, text=UCVOrange] {$\times$} (L0);

            % Cota inferior: la cadena completa es la profundidad disponible.
            \draw[<->, gray!70] ([yshift=-0.6cm]L3.south)
                -- node[below, font=\tiny, text=gray] {profundidad multiplicativa}
                   ([yshift=-0.6cm]L0.south);
        \end{tikzpicture}

\end{frame}

\begin{frame}{\textit{Bootstrapping}}
    El \textit{bootstrapping} evalúa homomórficamente la propia función de descifrado. Devuelve el texto cifrado con \textbf{ruido mínimo}, restaurando los niveles, y con ello habilita cómputos de profundidad indefinida.
    \vfill
    \begin{center}
        \begin{tikzpicture}[every node/.style={font=\scriptsize},
            cif/.style={draw=UCVBlue, fill=UCVBlue!8, rounded corners=3pt,
                        minimum width=1.35cm, minimum height=0.95cm},
            cifv/.style={draw=UCVAzul2, fill=UCVAzul2!8, rounded corners=3pt,
                         minimum width=1.35cm, minimum height=0.95cm},
            paso/.style={->, thick, gray!70}]

            % (A) Texto cifrado con la clave original: el ruido llega al umbral.
            \node[cif] (a) at (0,0) {$x$};
            \medidor{a}{0.89}\umbral{a}\candado{a}{UCVBlue}

            % (B) Se recifra bajo la clave nueva: la capa de fuera nace sin ruido.
            \node[cif, minimum width=1.15cm, minimum height=0.62cm, font=\tiny] (bi) at (2.5,-0.18) {$x$};
            \medidor{bi}{0.89}\umbral{bi}\candado{bi}{UCVBlue}
            \begin{scope}[on background layer]
                \node[cifv, fit=(bi)(bi-gl)(bi-gr), inner sep=9pt] (bo) {};
            \end{scope}
            \medidor{bo}{0.1}\candado{bo}{UCVAzul2}

            % (C) Se evalua el descifrado con la clave azul, cifrada bajo la verde.
            \node[draw=gray!70, fill=gray!8, rounded corners=3pt,
                  minimum width=2.5cm, minimum height=1.45cm] (c) at (5.6,0) {};
            \node[font=\tiny, text=gray!90, anchor=north] at ([yshift=-3pt]c.north) {Descifrado};
            \node[cif, minimum width=0.85cm, minimum height=0.48cm, font=\tiny]
                 (ci) at ([xshift=-7pt,yshift=-6pt]c.center) {$x$};
            \candado{ci}{UCVBlue}
            \node[font=\scriptsize, text=UCVBlue] at ([xshift=20pt]ci.east) {\faKey};

            % La llave de bootstrapping: la clave original cifrada con la nueva.
            \node[cifv, minimum width=1.25cm, minimum height=0.8cm] (bk) at (5.6,-2.2)
                 {\textcolor{UCVBlue}{\faKey}};
            \medidor{bk}{0.1}\candado{bk}{UCVAzul2}
            \draw[paso] ([yshift=2pt]$(bk-gl |- bk-gr)!0.5!(bk-gr)$) -- (c.south);
            \node[font=\tiny, text=UCVAzul2, align=right, anchor=east]
                 at ([xshift=-10pt]bk.west) {Llave de \textit{bootstrapping}\\ (pública)};

            % (D) Mismo dato, ahora bajo la clave nueva y con ruido bajo.
            \node[cifv] (d) at (8.5,0) {$x$};
            \medidor{d}{0.14}\umbral{d}\candado{d}{UCVAzul2}

            \draw[paso] (a.east)      -- (bo.west |- a);
            \draw[paso] (bo.east |- a) -- (c.west);
            \draw[paso] (c.east)      -- (d.west);
        \end{tikzpicture}
    \end{center}
    \vfill
\end{frame}

\begin{frame}{El esquema CKKS}
    Esquema de FHE para el cómputo sobre \textbf{números reales}. Trata el ruido como un error de redondeo, esto facilita al esquema el manejo de ruido durante operaciones.
    \vspace{0.35cm}
    \begin{listaalineada}
        \setlength{\itemsep}{10pt}
        \item \textbf{Reales:} Permite representar números reales sin usar escalado.
        \item \textbf{SIMD:} Múltiples números reales por texto cifrado pueden ser operados a la vez.
        \item \textbf{Operaciones:} Suma, multiplicación, rotación, relinearización y reescalado.
        \item \textbf{Bootstrapping:} Permite usar bootstrapping.
    \end{listaalineada}
\end{frame}

\begin{frame}{Objetivos}
    % El texto es literal del documento, asi que el ajuste es de tamaño y espaciado.
    \small
    El objetivo del trabajo es que OpalML facilite la inferencia privada de modelos de aprendizaje automático. Que sea una herramienta eficiente que combine la seguridad criptográfica con el rendimiento computacional necesario.
    \vspace{0.2cm}

    \textbf{Objetivos específicos}
    \vspace{0.1cm}
    \begin{listaalineada}
        \setlength{\itemsep}{4pt}
        \item Diseñar una interfaz en Python que permita interactuar con una biblioteca de cifrado homomórfico CKKS acelerada por GPU y emplear las operaciones básicas del esquema.
        \item Implementar las clases y módulos que permitan realizar operaciones con tensores y matrices, abstrayendo del usuario la complejidad del esquema CKKS.
        \item Crear casos de uso que ejemplifiquen el empleo de la biblioteca para aprovechar el cifrado homomórfico durante la inferencia en modelos de aprendizaje automático.
        \item Comparar el rendimiento de las redes neuronales implementadas con la biblioteca propuesta frente a implementaciones equivalentes construidas sobre otras bibliotecas de cifrado homomórfico.
    \end{listaalineada}
\end{frame}
